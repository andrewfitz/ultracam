#include <metal_stdlib>
using namespace metal;
// ABI: ten aligned 16-byte rows; mirrored by MonitorKernelParameters.swift.
struct Params {
 uint width,height,planes,format;
 uint outWidth,outHeight,flags,scopeMode;
 float threshold,widthIRE,opacity,focusThreshold;
 float noise,focusOpacity,gain,sourceLogRescaled;
 uint scopeWidth,scopeHeight,scopeSamplingStride,scopeSamplingPhase;
 // originX/originY: the viewport's top-left pixel in the target.
 float referenceOffsetStops,originX,originY,reserved2;
 float4 coefficients,mapping,peak;
 // Picture region (x, y, width, height in picture pixels) shown across the whole target.
 float4 aperture;
};
enum MonitorFlags : uint { Zebra=1u, ZebraBand=2u, Focus=4u, LogC3=8u, LogC4=16u,
 ScopeGraticule=1u, SonyARRIReference=128u, SonySGamut3=256u, SonySLog3=512u, ELZone=1024u };
typedef texture2d<uint,access::read> plane;
// Frames are read in place from their IOSurfaces:
//  0 v210: rgba32Uint, one texel per 6-pixel group
//  1 BGRA: rgba8Uint (x=B, y=G, z=R)
//  2,3 8-bit biplanar: r8Uint luma, rg8Uint CbCr
//  4..7 10-bit biplanar: r16Uint luma, rg16Uint CbCr, high-aligned
float3 readRGB(plane p0,plane p1,uint x,uint y,constant Params& p) {
 float Y,Cb,Cr;
 if(p.format==0) { uint4 g=p0.read(uint2(x/6,y)); uint a=g.x,c=g.y,d=g.z,e=g.w; uint n=x%6;
 uint ys[6]={(a>>10)&1023,c&1023,(c>>20)&1023,(d>>10)&1023,e&1023,(e>>20)&1023};
 uint us[3]={a&1023,(c>>10)&1023,(d>>20)&1023}; uint vs[3]={(a>>20)&1023,d&1023,(e>>10)&1023};
 bool full=p.coefficients.w>0; Y=(float(ys[n])-(full?0:64))/(full?1023:876); Cb=(float(us[n/2])-512)/(full?1023:896); Cr=(float(vs[n/2])-512)/(full?1023:896);
 } else if(p.format==1) { uint4 a=p0.read(uint2(x,y)); return float3(a.z,a.y,a.x)/255; }
 else { bool ten=p.format>=4; bool full=p.coefficients.w>0; uint2 c=uint2(x/2,p.format>=6?y:y/2); uint2 uv=p1.read(c).xy;
 if(ten) { Y=float(p0.read(uint2(x,y)).x>>6); Cb=float(uv.x>>6); Cr=float(uv.y>>6); Y=(Y-(full?0:64))/(full?1023:876); Cb=(Cb-512)/(full?1023:896); Cr=(Cr-512)/(full?1023:896); }
 else { Y=float(p0.read(uint2(x,y)).x); Cb=float(uv.x); Cr=float(uv.y); Y=(Y-(full?0:16))/(full?255:219); Cb=(Cb-128)/(full?255:224); Cr=(Cr-128)/(full?255:224); }
 }
 if(Cb==0&&Cr==0)return float3(Y);
 float r=Y+2*(1-p.coefficients.x)*Cr; float b=Y+2*(1-p.coefficients.z)*Cb; return float3(r,(Y-p.coefficients.x*r-p.coefficients.z*b)/p.coefficients.y,b);
}
float luma(float3 rgb){return rgb.g+.2126*(rgb.r-rgb.g)+.0722*(rgb.b-rgb.g); }
float signalLuma(float3 rgb,constant Params& p){return dot(rgb,p.coefficients.xyz);}
float sonySLog3Decode(float value){float code=value*1023;return code>=171.2102946929f?pow(10.0f,(code-420)/261.5f)*.19f-.01f:(code-95)*.01125f/(171.2102946929f-95);}
float sonyLogC3ReferenceCode(float value){return (value>.010591f?.247190f*log10(5.555556f*value+.052272f)+.385537f:5.367655f*value+.092809f)*1023;}

// Encoded transport Y equals declared matrix-weighted decoded RGB without
// reconstructing chroma. Peaking never uses a scene-linear/gamma transform.
float readSignalY(plane p0,plane p1,uint x,uint y,constant Params& p) {
 if(p.format==1)return signalLuma(readRGB(p0,p1,x,y,p),p);
 bool full=p.coefficients.w>0;
 if(p.format==0){
  uint4 g=p0.read(uint2(x/6,y));
  uint n=x%6,code;
  switch(n){case 0:code=(g.x>>10)&1023;break;
   case 1:code=g.y&1023;break;case 2:code=(g.y>>20)&1023;break;
   case 3:code=(g.z>>10)&1023;break;case 4:code=g.w&1023;break;
   default:code=(g.w>>20)&1023;break;}
  return (float(code)-(full?0:64))/(full?1023:876);
 }
 if(p.format>=4){float code=float(p0.read(uint2(x,y)).x>>6);return (code-(full?0:64))/(full?1023:876);}
 return (float(p0.read(uint2(x,y)).x)-(full?0:16))/(full?255:219);
}
float directionalMedian(float a,float b,float c){return max(min(a,b),min(max(a,b),c));}
float3 nativeSLog3(float3 rgb,constant Params& p){
 if(p.format==1)return p.sourceLogRescaled>0?(rgb*1023-64)/876:rgb;
 return p.coefficients.w>0||p.sourceLogRescaled>0?rgb:(rgb*876+64)/1023;
}
float3 sonyPattern2(float signal){
 float ire=(signal*1023-64)/876*100;
 float2 bounds[10]={float2(93.4,96.1),float2(87.7,90.6),float2(91.3,93.4),float2(43.8,46.5),float2(54.3,58),float2(47.8,50.8),float2(38.9,42.2),float2(24.6,34.4),float2(3.5,5.6),float2(-7,3.5)};
 // Sony names the color families; numerical RGB are Ultracam display choices.
 float3 colors[10]={float3(1,0,0),float3(1,.5,0),float3(1,1,0),float3(0,1,1),float3(1,.4,.7),float3(1,.7,.8),float3(0,1,0),float3(.3,.65,1),float3(0,0,1),float3(.6,0,1)};
 for(uint i=0;i<10;i++)if(ire>=bounds[i].x&&ire<bounds[i].y)return colors[i];
 return float3(signal);
}
// Public PanasonicFW9 chart palette plus Ultracam's explicitly versioned
// midpoint interval convention. This does not assert Sony M&C bit-exactness.
float3 elZonePublicChart(float3 native,constant Params& p){
 float3 linear=float3(sonySLog3Decode(native.r),sonySLog3Decode(native.g),sonySLog3Decode(native.b));
 float3 weights=(p.flags&SonySGamut3)?float3(.2709796708f,.7866064112f,-.0575860820f):float3(.215075820115587f,.885068501743727f,-.100144321859314f);
 float Y=dot(linear,weights);
 if(Y<=0)return float3(0);
 if(!isfinite(Y))return float3(luma(native));
 float stops=log2(Y)-log2(.18f)+p.referenceOffsetStops;
 if(stops<=-6)return float3(0);
 float upper[13]={-4.5,-3.5,-2.5,-1.5,-.75,-.25,.25,.75,1.5,2.5,3.5,4.5,6};
 float3 colors[15]={float3(0),float3(.439,.188,.627),float3(0,.439,.753),float3(0,.69,.941),float3(.0824,.651,.0549),float3(.192,.878,.141),float3(0,1,0),float3(.533,.537,.506),float3(.973,.949,0),float3(1,.992,.463),float3(1,.565,0),float3(1,.694,.357),float3(1,0,0),float3(1,.412,.412),float3(1)};
 for(uint i=0;i<13;i++)if(stops<upper[i])return colors[i+1];
 return colors[14];
}
float3 bilinearRGB(plane p0,plane p1,float2 point,constant Params& p){
 float2 pt=clamp(point,float2(0),float2(p.width-1,p.height-1));
 uint2 lo=uint2(floor(pt)),hi=min(lo+1,uint2(p.width-1,p.height-1));float2 f=fract(pt);
 return mix(mix(readRGB(p0,p1,lo.x,lo.y,p),readRGB(p0,p1,hi.x,lo.y,p),f.x),
  mix(readRGB(p0,p1,lo.x,hi.y,p),readRGB(p0,p1,hi.x,hi.y,p),f.x),f.y);
}
float2 pictureScale(constant Params& p){return p.aperture.zw/float2(p.outWidth,p.outHeight);}
// Identity is exact. Integer minification <=8 is box averaged; fractional
// ratios use bounded stratified bilinear quadrature (max8x8 taps). This is not
// a calibrated chroma-siting transform; identity chroma decoding is unchanged.
float3 filteredPicture(plane p0,plane p1,uint2 xy,constant Params& p){
 float2 scale=pictureScale(p);
 bool integralOrigin=all(p.aperture.xy==floor(p.aperture.xy));
 if(all(scale==float2(1))&&integralOrigin){
  uint2 at=min(xy+uint2(p.aperture.xy),uint2(p.width-1,p.height-1));
  return readRGB(p0,p1,at.x,at.y,p);
 }
 uint2 count=uint2(clamp(ceil(scale),float2(1),float2(8)));
 float3 sum=0;bool integral=integralOrigin&&all(scale==floor(scale))&&all(scale<=8);
 for(uint y=0;y<count.y;y++)for(uint x=0;x<count.x;x++){
  float2 pos=p.aperture.xy+(float2(xy)+(float2(x,y)+.5)/float2(count))*scale-.5;
  if(integral){uint2 at=uint2(clamp(round(pos),float2(0),float2(p.width-1,p.height-1)));sum+=readRGB(p0,p1,at.x,at.y,p);}
  else sum+=bilinearRGB(p0,p1,pos,p);
 }
 return sum/float(count.x*count.y);
}
float focusFootprint(texture2d<float,access::read> mask,uint2 xy,constant Params& p){
 float2 scale=pictureScale(p);
 float2 a=(p.aperture.xy+float2(xy)*scale-p.mapping.zw)/p.mapping.xy;
 float2 b=(p.aperture.xy+(float2(xy)+1)*scale-p.mapping.zw)/p.mapping.xy;
 float2 low=min(a,b),high=max(a,b);
 float2 extent=ceil(high)-floor(low);
 uint2 count=uint2(clamp(extent,float2(1),float2(8)));
 bool subsampled=any(extent>8);
 float confidence=0;
 for(uint y=0;y<count.y;y++)for(uint x=0;x<count.x;x++){
  float2 pos=subsampled?floor(low+(float2(x,y)+.5)/float2(count)*(high-low)):floor(low)+float2(x,y);
  uint2 at=uint2(clamp(pos,float2(0),float2(mask.get_width()-1,mask.get_height()-1)));
  confidence=max(confidence,mask.read(at).r);
 }
 return confidence;
}
// Source-resolution peaking mask. All threads participate in the clamped halo
// load, including padded threads; xy-localID is the tile origin.
kernel void sourceFocus(plane p0 [[texture(0)]],plane p1 [[texture(1)]],constant Params& p [[buffer(0)]],texture2d<float,access::write> mask [[texture(2)]],uint2 xy [[thread_position_in_grid]],uint2 localID [[thread_position_in_threadgroup]],uint2 groupSize [[threads_per_threadgroup]]) {
 threadgroup float tile[18*18];
 float tl,tc,tr,ml,mr,bl,bc,br;
 if(groupSize.x<=16&&groupSize.y<=16){
  uint pitch=groupSize.x+2,tileCount=pitch*(groupSize.y+2),threads=groupSize.x*groupSize.y;
  uint lane=localID.y*groupSize.x+localID.x;
  int2 origin=int2(xy)-int2(localID);
  for(uint i=lane;i<tileCount;i+=threads){
   int2 position=clamp(origin+int2(i%pitch,i/pitch)-1,int2(0),int2(p.width-1,p.height-1));
   tile[i]=readSignalY(p0,p1,uint(position.x),uint(position.y),p);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);
  if(xy.x>=p.width||xy.y>=p.height)return;
  uint center=(localID.y+1)*pitch+localID.x+1;
  tl=tile[center-pitch-1];tc=tile[center-pitch];tr=tile[center-pitch+1];
  ml=tile[center-1];mr=tile[center+1];
  bl=tile[center+pitch-1];bc=tile[center+pitch];br=tile[center+pitch+1];
 }else{
  // Keep other legal dispatch sizes supported without an undersized tile.
  if(xy.x>=p.width||xy.y>=p.height)return;
  uint l=xy.x?xy.x-1:0,r=min(xy.x+1,p.width-1),u=xy.y?xy.y-1:0,d=min(xy.y+1,p.height-1);
  tl=readSignalY(p0,p1,l,u,p);tc=readSignalY(p0,p1,xy.x,u,p);tr=readSignalY(p0,p1,r,u,p);
  ml=readSignalY(p0,p1,l,xy.y,p);mr=readSignalY(p0,p1,r,xy.y,p);
  bl=readSignalY(p0,p1,l,d,p);bc=readSignalY(p0,p1,xy.x,d,p);br=readSignalY(p0,p1,r,d,p);
 }
 float3 dx=float3(tr-tl,mr-ml,br-bl),dy=float3(bl-tl,bc-tc,br-tr);
 float gx=dot(dx,float3(3,10,3))/16,gy=dot(dy,float3(3,10,3))/16;
 float support=max(abs(directionalMedian(dx.x,dx.y,dx.z)),abs(directionalMedian(dy.x,dy.y,dy.z)));
 float confidence=support>p.noise?clamp((max(abs(gx),abs(gy))-p.focusThreshold)/max(.0001f,p.focusThreshold*.5f),0.0f,1.0f):0;
 mask.write(float4(confidence),xy);
}
// The picture: one pass from the frame's own planes into the display target.
kernel void compose(plane p0 [[texture(0)]],plane p1 [[texture(1)]],plane s0 [[texture(2)]],plane s1 [[texture(3)]],constant Params& p [[buffer(0)]],constant Params& sp [[buffer(1)]],texture2d<float,access::read> focus [[texture(4)]],texture2d<float,access::write> output [[texture(5)]],uint2 xy [[thread_position_in_grid]]) {
 if(xy.x>=p.outWidth||xy.y>=p.outHeight)return;
 float2 center=p.aperture.xy+(float2(xy)+.5)*pictureScale(p)-.5;
 bool falseColor=(p.flags&(LogC3|LogC4|SonySLog3|ELZone))!=0;
 // Source classification replaces the picture: do not minify a look we discard.
 float3 c=falseColor?float3(0):filteredPicture(p0,p1,xy,p);
 if(falseColor){
  uint2 sourceXY=uint2(clamp((center+.5-p.mapping.zw)/p.mapping.xy-.5,float2(0),float2(sp.width-1,sp.height-1)));
  if(p.flags&ELZone){c=elZonePublicChart(nativeSLog3(readRGB(s0,s1,sourceXY.x,sourceXY.y,sp),sp),p);}
  else if(p.flags&SonySLog3){float3 native=nativeSLog3(readRGB(s0,s1,sourceXY.x,sourceXY.y,sp),sp);c=sonyPattern2(signalLuma(native,sp));}
  else {
 float3 nativeRGB=readRGB(s0,s1,sourceXY.x,sourceXY.y,sp);float a=luma(nativeRGB);bool full=sp.coefficients.w>0||sp.sourceLogRescaled>0;float code=a*(full?1023:876)+(full?0:64);
 // Neutral native10bit samples have integral native code. Restore that exact
 // integer after float normalization; never quantize colored or rescaled input.
 if((sp.format==0||sp.format>=4)&&sp.sourceLogRescaled==0&&nativeRGB.r==nativeRGB.g&&nativeRGB.g==nativeRGB.b)code=round(code);
 if(p.flags&SonyARRIReference){
 // YUV has already been range-expanded by readRGB; BGRA has not. Undo only
 // that container interpretation to recover original normalized native S-Log3.
 float3 slog3=sp.format==1?(sp.sourceLogRescaled>0?(nativeRGB*1023-64)/876:nativeRGB):(full?nativeRGB:(nativeRGB*876+64)/1023);
 float3 linear=float3(sonySLog3Decode(slog3.r),sonySLog3Decode(slog3.g),sonySLog3Decode(slog3.b));
 float3 weights=(p.flags&SonySGamut3)?float3(.2709796708f,.7866064112f,-.0575860820f):float3(.215075820115587f,.885068501743727f,-.100144321859314f);
 code=sonyLogC3ReferenceCode(dot(linear,weights));a=luma(slog3);full=true;
 }
 float2 z3legal[6]={float2(878,940),float2(857,878),float2(461,480),float2(397,415),float2(152,172),float2(64,152)};
 float2 z3full[6]={float2(951,1023),float2(926,951),float2(464,486),float2(389,410),float2(103,127),float2(0,103)};
 float2 z4legal[6]={float2(808,940),float2(789,808),float2(351,366),float2(300,315),float2(149,159),float2(64,149)};
 float2 z4full[6]={float2(868,1023),float2(846,868),float2(335,353),float2(276,293),float2(99,111),float2(0,99)};
 float3 colors[6]={float3(1,0,0),float3(1,1,0),float3(1,.7,.7),float3(0,1,0),float3(0,0,1),float3(.7,0,1)};c=float3(a);
 for(uint i=0;i<6;i++){float2 z=(p.flags&LogC3)?(full?z3full[i]:z3legal[i]):(full?z4full[i]:z4legal[i]);if(code>=z.x&&code<z.y){c=colors[i];break;}}
  }
 }
 float ire=signalLuma(c,p)*100;
 bool zebra=(p.flags&Zebra)&&((p.flags&ZebraBand)?(ire>=p.threshold-p.widthIRE*.5&&ire<p.threshold+p.widthIRE*.5):(ire>=p.threshold));
 if(zebra&&((xy.x+xy.y)%16<8))c=mix(c,float3(1),p.opacity);
 if(p.flags&Focus)c=mix(c,p.peak.rgb,p.focusOpacity*focusFootprint(focus,xy,p));
 output.write(float4(clamp(c,0.0f,1.0f),1),xy+uint2(p.originX,p.originY));
}
// Black, for a surface with nothing to show.
kernel void clearTarget(texture2d<float,access::write> output [[texture(0)]],uint2 xy [[thread_position_in_grid]]){
 if(xy.x<output.get_width()&&xy.y<output.get_height())output.write(float4(0,0,0,1),xy);
}
// Exact population of a sampled lattice inside a half-open source interval.
uint latticeCount(uint low,uint high,uint offset,uint step){
 uint begin=low>offset?(low-offset+step-1)/step:0;
 uint end=high>offset?(high-offset+step-1)/step:0;
 return end>begin?end-begin:0;
}
// Source columns [low, high) that land in scope column `column` of `columns`.
uint2 scopeColumnSpan(uint column,uint columns,uint width){
 return uint2((column*width+columns-1)/columns,((column+1)*width+columns-1)/columns);
}
// Density population depends on x and uniform sampling metadata, never y.
uint scopeColumnPopulation(uint x,constant Params& p){
 uint step=max(1u,p.scopeSamplingStride),phase=p.scopeSamplingPhase%(step*step);
 uint columns=p.scopeMode==1?p.scopeWidth:p.scopeWidth/3,column=x%columns;
 uint2 span=scopeColumnSpan(column,columns,p.width);
 return latticeCount(span.x,span.y,phase%step,step)*latticeCount(0,p.height,phase/step,step);
}
uint scopeBin(float value,constant Params& p){return uint(clamp((value+.1f)/1.2f,0.0f,1.0f)*float(p.scopeHeight-1));}
float3 scopeTexel(uint x,uint y,float n,constant Params& p){
 uint columns=p.scopeMode==1?p.scopeWidth:p.scopeWidth/3;
 uint population=scopeColumnPopulation(x,p);
 float v=population?1-exp(-n*p.gain*30.72f/float(population)):0;
 float3 color=p.scopeMode==1?float3(1):x<columns?float3(1,.1,.1):x<2*columns?float3(.1,1,.1):float3(.1,.3,1);
 float3 rendered=color*v;
 if(p.flags&ScopeGraticule){bool horizontal=y==0||y==21||y==128||y==234||y==255;bool vertical=p.scopeMode==1?x%192==0:x%64==0;if(horizontal||vertical)rendered=max(rendered,float3(.16));}
 return rendered;
}
// Waveform / RGB parade in one pass, no device atomics or bins buffer. One
// threadgroup per scope column accumulates every sampled pixel of its source
// strip into threadgroup bins, then writes that column of the scope image.
// Parade columns hold all three channels (scopeWidth/3 groups, three output
// columns each). Dispatch scopeWidth (luma) or scopeWidth/3 (parade) groups of
// scopeHeight threads; scopeHeight must be <= 256.
kernel void scopeColumns(plane p0 [[texture(0)]],plane p1 [[texture(1)]],constant Params& p [[buffer(0)]],texture2d<float,access::write> output [[texture(2)]],uint group [[threadgroup_position_in_grid]],uint lane [[thread_index_in_threadgroup]],uint threads [[threads_per_threadgroup]]) {
 threadgroup atomic_uint bins[3*256];
 bool parade=p.scopeMode!=1;
 uint channels=parade?3:1,columns=parade?p.scopeWidth/3:p.scopeWidth;
 for(uint i=lane;i<channels*256;i+=threads)atomic_store_explicit(&bins[i],0,memory_order_relaxed);
 threadgroup_barrier(mem_flags::mem_threadgroup);
 uint step=max(1u,p.scopeSamplingStride),phase=p.scopeSamplingPhase%(step*step);
 uint2 span=scopeColumnSpan(group,columns,p.width);
 uint offsetX=phase%step,offsetY=phase/step;
 uint firstX=span.x>offsetX?((span.x-offsetX+step-1)/step)*step+offsetX:offsetX;
 uint stripWidth=firstX<span.y?(span.y-firstX+step-1)/step:0;
 uint rows=p.height>offsetY?(p.height-offsetY+step-1)/step:0;
 uint total=stripWidth*rows;
 for(uint i=lane;i<total;i+=threads){
  uint x=firstX+(i%stripWidth)*step,y=offsetY+(i/stripWidth)*step;
  float3 rgb=readRGB(p0,p1,x,y,p);
  if(parade){for(uint c=0;c<3;c++)atomic_fetch_add_explicit(&bins[c*256+scopeBin(rgb[c],p)],1,memory_order_relaxed);}
  else atomic_fetch_add_explicit(&bins[scopeBin(signalLuma(rgb,p),p)],1,memory_order_relaxed);
 }
 threadgroup_barrier(mem_flags::mem_threadgroup);
 for(uint row=lane;row<p.scopeHeight;row+=threads){
  uint bin=p.scopeHeight-1-row;
  for(uint c=0;c<channels;c++){
   uint x=c*columns+group;
   float n=float(atomic_load_explicit(&bins[c*256+bin],memory_order_relaxed));
   output.write(float4(scopeTexel(x,row,n,p),1),uint2(x,row));
  }
 }
}
